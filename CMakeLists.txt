cmake_minimum_required(VERSION 3.20)
project(DesktopBox VERSION 0.4.7 LANGUAGES CXX RC)
option(DESKTOPBOX_BUILD_TESTS "Build the separate native regression-test executable" OFF)
set(DESKTOPBOX_SOURCES src/main.cpp src/model.cpp src/selection.cpp src/storage.cpp src/desktop.cpp src/desktop_mouse.cpp src/native_entry.cpp src/backdrop.cpp src/shell_menu.cpp src/shell_registration.cpp src/fluent_settings.cpp resources/app.rc)
add_executable(DesktopBox WIN32 ${DESKTOPBOX_SOURCES})
function(configure_desktopbox target)
target_compile_features(${target} PRIVATE cxx_std_17)
target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(${target} PRIVATE user32 gdi32 shell32 shlwapi ole32 oleaut32 uuid comdlg32 comctl32 dwmapi advapi32 taskschd uxtheme gdiplus)
if(MSVC)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/Gy> $<$<CONFIG:Release>:/Gw>)
    target_link_options(${target} PRIVATE /MANIFEST:NO $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
else()
    target_compile_options(${target} PRIVATE -Wall -Wextra $<$<CONFIG:Release>:-Os>)
    target_link_options(${target} PRIVATE -municode)
endif()
endfunction()
configure_desktopbox(DesktopBox)
if(DESKTOPBOX_BUILD_TESTS)
    add_executable(DesktopBoxTests WIN32 ${DESKTOPBOX_SOURCES})
    set_target_properties(DesktopBoxTests PROPERTIES OUTPUT_NAME DesktopBox-tests)
    configure_desktopbox(DesktopBoxTests)
    target_compile_definitions(DesktopBoxTests PRIVATE DESKTOPBOX_TESTS)
endif()
